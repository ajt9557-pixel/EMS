1. AUTHENTICATION MODULE

FUNCTION LOGIN(email, password, rememberMe):
    IF email == "" OR password == "":
        SHOW ERROR "Please fill in all fields"
        RETURN

    SET loading = true
    SHOW LoadingOverlay(logo, spinner)

    response = POST /api/auth/login { email, password }

    IF response.success == true:
        token = response.token
        user  = response.user

        IF rememberMe == true:
            SAVE token to localStorage
        ELSE:
            SAVE token to sessionStorage

        SAVE user to AuthContext (global state)

        IF user.role == "admin":
            NAVIGATE to /admin-dashboard
        ELSE:
            NAVIGATE to /employee-dashboard

    ELSE:
        SHOW ERROR response.error

    SET loading = false
    HIDE LoadingOverlay

FUNCTION REGISTER(name, email, password):
    VALIDATE all fields are filled
    response = POST /api/auth/register { name, email, password }
    IF response.success: NAVIGATE to /login

FUNCTION LOGOUT():
    DELETE token from localStorage AND sessionStorage
    CLEAR user from AuthContext
    NAVIGATE to /login

FUNCTION VERIFY_SESSION():
    token = GET from localStorage OR sessionStorage
    IF token exists:
        response = GET /api/auth/verify (with Bearer token)
        IF response.success: SET user in AuthContext
        ELSE: LOGOUT()

2. ROUTING / ACCESS CONTROL
ROUTES:
    "/"                        -> redirect to /login
    "/login"                   -> Login page (public)
    "/register"                -> Register page (public)
    "/admin-dashboard/*"       -> Protected (admin only)
    "/employee-dashboard/*"    -> Protected (employee only)

MIDDLEWARE PRIVATE_ROUTE(component):
    IF no token: NAVIGATE to /login
    ELSE: RENDER component

MIDDLEWARE ROLE_BASE_ROUTE(component, requiredRoles):
    IF user.role NOT IN requiredRoles:
        SHOW "Access Denied" page
    ELSE:
        RENDER component

5. EMPLOYEE MODULE (CRUD)

FUNCTION VIEW_EMPLOYEES(searchKeyword):
    IF searchKeyword != "":
        employees = GET /api/employees?search=searchKeyword
    ELSE:
        employees = GET /api/employees

    DISPLAY table/cards:
        - Profile picture, Name, Email, Department, Role, Status, Actions
        - Search bar (filters as you type)
        - Button "Add New Employee"

FUNCTION ADD_EMPLOYEE(data):
    VALIDATE required fields (name, email, department, role)
    response = POST /api/employees { data }
    IF success: REFRESH list, SHOW "Employee added"
    ELSE: SHOW error

FUNCTION EDIT_EMPLOYEE(id, updatedData):
    response = PUT /api/employees/id { updatedData }
    IF success: REFRESH list, SHOW "Employee updated"

FUNCTION DELETE_EMPLOYEE(id):
    CONFIRM "Are you sure?"
    response = DELETE /api/employees/id
    IF success: REMOVE from list, SHOW "Employee deleted"
